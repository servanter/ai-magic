## Setup: Three Approaches

### Approach 1: Stable Diffusion Locally (Free, Full Control)

**Pros**: No API costs, full model customization, no rate limits  
**Cons**: Requires GPU (8GB+ VRAM), slower than cloud APIs  

**Installation**:

```python
# Install dependencies
pip install diffusers transformers accelerate torch pillow

# Verify GPU availability
import torch
print(f"CUDA available: {torch.cuda.is_available()}")
print(f"GPU: {torch.cuda.get_device_name(0) if torch.cuda.is_available() else 'None'}")
```

**Basic Usage**:

```python
from diffusers import StableDiffusionPipeline
import torch

# Load model (Stable Diffusion XL is recommended in 2026)
model_id = "stabilityai/stable-diffusion-xl-base-1.0"
pipe = StableDiffusionPipeline.from_pretrained(
    model_id,
    torch_dtype=torch.float16,  # Use FP16 for speed
    use_safetensors=True
)
pipe = pipe.to("cuda")

# Generate image
prompt = "A serene mountain landscape at sunset, photorealistic, 8k quality"
image = pipe(
    prompt,
    num_inference_steps=30,  # 20-50 steps typical
    guidance_scale=7.5,      # Higher = closer to prompt
    height=1024,
    width=1024
).images[0]

image.save("output.jpg")
```

**Performance Tips**:
- Use `torch.float16` (FP16) for 2x speed improvement
- Enable `xformers` for memory efficiency: `pipe.enable_xformers_memory_efficient_attention()`
- Batch multiple prompts: `pipe(["prompt1", "prompt2", "prompt3"])`

## Building a Production API

Here's a FastAPI-based image generation service that supports multiple backends:

```python
from fastapi import FastAPI, HTTPException, BackgroundTasks
from pydantic import BaseModel
from typing import Literal
import uvicorn
from PIL import Image
import io
import base64
import asyncio

app = FastAPI(title="AI Image Generator API")

class GenerateRequest(BaseModel):
    prompt: str
    model: Literal["stable-diffusion", "dalle-3", "midjourney"] = "stable-diffusion"
    width: int = 1024
    height: int = 1024
    num_steps: int = 30
    guidance_scale: float = 7.5

class GenerateResponse(BaseModel):
    image_base64: str
    model_used: str
    generation_time_seconds: float
    cost_estimate: float

# Initialize models (lazy loading)
sd_pipe = None

def load_stable_diffusion():
    global sd_pipe
    if sd_pipe is None:
        from diffusers import StableDiffusionPipeline
        import torch
        sd_pipe = StableDiffusionPipeline.from_pretrained(
            "stabilityai/stable-diffusion-xl-base-1.0",
            torch_dtype=torch.float16
        ).to("cuda")
        sd_pipe.enable_xformers_memory_efficient_attention()
    return sd_pipe

@app.post("/generate", response_model=GenerateResponse)
async def generate_image(request: GenerateRequest):
    import time
    start_time = time.time()
    
    try:
        if request.model == "stable-diffusion":
            pipe = load_stable_diffusion()
            image = pipe(
                request.prompt,
                num_inference_steps=request.num_steps,
                guidance_scale=request.guidance_scale,
                height=request.height,
                width=request.width
            ).images[0]
            cost_estimate = 0.0  # Free (excluding GPU costs)
        
        elif request.model == "dalle-3":
            from openai import OpenAI
            import requests
            client = OpenAI()
            response = client.images.generate(
                model="dall-e-3",
                prompt=request.prompt,
                size=f"{request.width}x{request.height}",
                quality="hd"
            )
            image_response = requests.get(response.data[0].url)
            image = Image.open(io.BytesIO(image_response.content))
            cost_estimate = 0.08  # HD 1024x1024 pricing
        
        elif request.model == "midjourney":
            import replicate
            output = replicate.run(
                "stability-ai/sdxl:39ed52f2a78e934b3ba6e2a89f5b1c712de7dfea535525255b1aa35c5565e08b",
                input={
                    "prompt": request.prompt,
                    "width": request.width,
                    "height": request.height,
                    "num_inference_steps": request.num_steps
                }
            )
            image_response = requests.get(output[0])
            image = Image.open(io.BytesIO(image_response.content))
            cost_estimate = 0.015  # Approximate
        
        # Convert to base64
        buffered = io.BytesIO()
        image.save(buffered, format="JPEG", quality=95)
        img_base64 = base64.b64encode(buffered.getvalue()).decode()
        
        generation_time = time.time() - start_time
        
        return GenerateResponse(
            image_base64=img_base64,
            model_used=request.model,
            generation_time_seconds=round(generation_time, 2),
            cost_estimate=cost_estimate
        )
    
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.get("/health")
async def health_check():
    return {"status": "ok", "models": ["stable-diffusion", "dalle-3", "midjourney"]}

if __name__ == "__main__":
    uvicorn.run(app, host="0.0.0.0", port=8000)
```

**Usage**:


```python
    "runwayml/stable-diffusion-v1-5",

# Load control image (e.g., edge-detected photo)
```

## Performance Optimization

### 1. GPU Memory Management

```python
import torch
from diffusers import StableDiffusionPipeline

# Enable CPU offloading for large models
pipe = StableDiffusionPipeline.from_pretrained(
    "stabilityai/stable-diffusion-xl-base-1.0",
    torch_dtype=torch.float16
)
pipe.enable_model_cpu_offload()  # Move layers to CPU when not in use

# Or use sequential CPU offload (slower but uses less VRAM)
pipe.enable_sequential_cpu_offload()

# Clear cache between generations
torch.cuda.empty_cache()
```

### 2. Batch Processing

```python
# Generate multiple images in parallel
prompts = [
    "A sunset over mountains",
    "A bustling city street at night",
    "A serene beach with palm trees"
]

images = pipe(
    prompts,

    height=1024,
    width=1024

for i, img in enumerate(images):
    img.save(f"batch_{i}.jpg")
```

**Throughput Comparison**:
- Sequential: 3 images × 20s = 60s
- Batch: 3 images in 35s (43% faster)

- Copyrighted characters or brands (Mickey Mouse, Superman)
- Public figures without context
- Adult content

**Mitigation**:
- Pre-filter prompts with keyword blacklist
- Use prompt moderation API (OpenAI Moderation API)
- Provide user guidelines

### 2. Rate Limit Handling

```python
import time
from openai import RateLimitError

def generate_with_retry(client, prompt, max_retries=3):
    for attempt in range(max_retries):
        try:
            response = client.images.generate(
                model="dall-e-3",
                prompt=prompt
            )
            return response
        except RateLimitError as e:
            if attempt < max_retries - 1:
                wait_time = 2 ** attempt  # Exponential backoff: 1s, 2s, 4s
                print(f"⏳ Rate limited. Retrying in {wait_time}s...")
                time.sleep(wait_time)
            else:
                raise
```

### 3. Timeout Protection

```python
import asyncio

async def generate_with_timeout(prompt, timeout_seconds=60):
    try:
        result = await asyncio.wait_for(
            asyncio.to_thread(pipe, prompt, num_inference_steps=30),
            timeout=timeout_seconds
        )
        return result.images[0]
    except asyncio.TimeoutError:
        print("❌ Generation timeout. Using fallback image.")
        return Image.open("fallback.jpg")
```

### Strategy 2: Caching Popular Prompts

```python
import hashlib
import os

def generate_with_cache(prompt, cache_dir="cache"):
    # Generate cache key
    prompt_hash = hashlib.md5(prompt.encode()).hexdigest()
    cache_path = f"{cache_dir}/{prompt_hash}.jpg"
    
    # Check cache
    if os.path.exists(cache_path):
        print("✅ Cache hit")
        return Image.open(cache_path)
    
    # Generate and cache
    image = pipe(prompt).images[0]
    os.makedirs(cache_dir, exist_ok=True)
    image.save(cache_path)
    return image
```

**Impact**: 30-40% cache hit rate for generic prompts (e.g., "A mountain landscape").

### Issue 1: Out of Memory (OOM)

**Error**: `RuntimeError: CUDA out of memory`

**Solutions**:
```python
# 1. Reduce resolution
image = pipe(prompt, height=512, width=512)  # Instead of 1024x1024

# 2. Use FP16
pipe = pipe.to("cuda", torch_dtype=torch.float16)

# 3. Enable CPU offloading
pipe.enable_sequential_cpu_offload()

# 4. Clear cache
import torch
torch.cuda.empty_cache()
```

### Issue 2: Slow Generation (Greater than 60s)

**Diagnosis**:
```python
import time

start = time.time()
image = pipe(prompt, num_inference_steps=50)
print(f"Generation took {time.time() - start:.2f}s")
```

**Solutions**:
- Reduce `num_inference_steps` (30 is usually sufficient)
- Enable `xformers`: `pipe.enable_xformers_memory_efficient_attention()`
- Use smaller model (SD 1.5 instead of XL)
- Upgrade GPU (RTX 3060 → RTX 4090 = 3x faster)

### Issue 3: Poor Quality Results

**Common Causes**:
1. **Vague prompts** → Add specific details (style, lighting, composition)
2. **Low steps** → Increase to 30-50
3. **Wrong guidance scale** → Try 7-9 for most cases
4. **Model mismatch** → Use SDXL for photorealism, SD 1.5 for art

**Prompt Engineering Tips**:
```python
# Bad prompt
"A cat"

# Good prompt
"A fluffy orange tabby cat sitting on a windowsill, golden hour lighting, photorealistic, 8k, detailed fur texture"

# Great prompt (with negative prompt)
positive_prompt = "A fluffy orange tabby cat sitting on a windowsill, golden hour lighting, photorealistic, 8k, detailed fur texture, bokeh background"
negative_prompt = "blurry, distorted, low quality, cartoon, anime, watermark"

    positive_prompt,

    num_inference_steps=40,
    guidance_scale=8.0
```
